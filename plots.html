Cool Plots

data("ny_noaa") #load dataset into environment
noaa_df <- ny_noaa

noaa_df = 
  noaa_df |>
#This creates separate variables for year, month and day and converts observations of temperature, precipitation and snowfall into more convenient scales (from mm or tenths of mm to cm and from tenths of degrees celsius to degrees celsius) 
  mutate(date = ymd(date),        
         year = year(date),
         month = month(date, label = TRUE),
         day = day(date),
         tmax_1deg = as.numeric(tmax)/10, #1C = 10 tenths of degrees celsius
         tmin_1deg = as.numeric(tmin)/10, #1C = 10 tenths of degrees celsius
         prcp_cm = prcp/100, #1 cm = 100 tenths of mm
         snow_cm = snow/10, #1 cm = 10 mm
         snwd_cm = snwd/10 #1 cm = 10 mm 
) |>
  filter(id == "USC00300668",
         !is.na(id)
  )
